cmake_minimum_required(VERSION 3.20)
project(kded-tray-standin LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Core DBus)
find_package(KF6CoreAddons REQUIRED)
find_package(KF6DBusAddons REQUIRED)
set(CMAKE_AUTOMOC ON)
add_library(statusnotifierwatcher MODULE standin.cpp)
set_target_properties(statusnotifierwatcher PROPERTIES PREFIX "")
target_link_libraries(statusnotifierwatcher Qt6::Core Qt6::DBus KF6::CoreAddons KF6::DBusAddons)
install(TARGETS statusnotifierwatcher DESTINATION $ENV{HOME}/.local/lib/qt6/plugins/kf6/kded)
